<script>
    // Esta función lee el CSV que exportó tu código Python
    $(document).ready(function() {
        // IMPORTANTE: Asegúrate de subir tu 'reporte_dinamico.csv' a la misma carpeta de GitHub
        fetch('reporte_dinamico.csv')
            .then(response => response.text())
            .then(data => {
                const filas = data.split('\n').slice(1); // Ignorar el encabezado
                let html = '';
                
                filas.forEach(fila => {
                    const columnas = fila.match(/(".*?"|[^",\s]+)(?=\s*,|\s*$)/g);
                    if (columnas && columnas.length >= 9) {
                        html += `<tr>`;
                        columnas.forEach(col => {
                            let textoLimpio = col.replace(/^"|"$/g, ''); // Quitar comillas
                            html += `<td>${textoLimpio}</td>`;
                        });
                        html += `</tr>`;
                    }
                });
                
                $('#cuerpoTabla').html(html);
                
                // Activar las funciones interactivas (Filtros, ordenamiento, páginas)
                $('#tablaDatos').DataTable({
                    language: { url: '//cdn.datatables.net/plug-ins/1.13.4/i18n/es-ES.json' },
                    order: [[0, 'desc']] // Ordenar para que aparezcan primero las categorías
                });
            })
            .catch(error => console.error('Error al cargar el CSV:', error));
    });
</script>
